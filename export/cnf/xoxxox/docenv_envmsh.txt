FROM ubuntu:24.04

ENV OPTAPT=""
#ENV OPTAPT="-o Acquire::AllowInsecureRepositories=true -o Acquire::AllowDowngradeToInsecureRepositories=true"
ENV DEBIAN_FRONTEND=noninteractive
RUN \
  apt-get ${OPTAPT} update && \
  apt-get ${OPTAPT} install -y --no-install-recommends python3 python3-venv && \
  rm -rf /var/lib/apt/lists/*
ENV DIRPYT=/env/python
RUN python3 -m venv ${DIRPYT}
RUN ${DIRPYT}/bin/pip3 install --no-cache-dir moonshine-voice aiohttp
